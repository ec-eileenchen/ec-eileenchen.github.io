1. Advanced Data Preprocessing Pipeline

Innovation: Custom preprocessing pipeline handling mixed data types, missing values, and feature engineering with memory optimization for large datasets.


import pandas as pd
import numpy as np
from sklearn.preprocessing import StandardScaler, LabelEncoder
from sklearn.model_selection import TimeSeriesSplit

class FraudDataPreprocessor:
    def __init__(self):
        self.scalers = {}
        self.encoders = {}
        self.feature_columns = []
    
    def merge_datasets(self, transaction_df, identity_df):
        """Merge transaction and identity data with memory optimization"""
        # Reduce memory usage by downcasting numeric types
        transaction_df = self.reduce_memory_usage(transaction_df)
        identity_df = self.reduce_memory_usage(identity_df)
        
        # Merge on TransactionID with indicator for missing identity data
        merged_df = pd.merge(
            transaction_df, 
            identity_df, 
            on='TransactionID', 
            how='left',
            indicator='has_identity_data'
        )
        return merged_df
    
    def reduce_memory_usage(self, df):
        """Aggressive memory optimization for large datasets"""
        for col in df.columns:
            col_type = df[col].dtype
            
            if col_type != object:
                c_min = df[col].min()
                c_max = df[col].max()
                
                if str(col_type)[:3] == 'int':
                    if c_min > np.iinfo(np.int8).min and c_max < np.iinfo(np.int8).max:
                        df[col] = df[col].astype(np.int8)
                    elif c_min > np.iinfo(np.int16).min and c_max < np.iinfo(np.int16).max:
                        df[col] = df[col].astype(np.int16)
                    # ... continue for int32, int64
                else:
                    if c_min > np.finfo(np.float16).min and c_max < np.finfo(np.float16).max:
                        df[col] = df[col].astype(np.float16)
                    # ... continue for float32
        return df
    
    def handle_missing_values(self, df):
        """Strategic missing value imputation based on feature analysis"""
        # For Vesta features, use median imputation
        vesta_cols = [col for col in df.columns if col.startswith('V')]
        df[vesta_cols] = df[vesta_cols].fillna(df[vesta_cols].median())
        
        # For categorical features, use mode or 'Missing' category
        cat_cols = df.select_dtypes(include=['object']).columns
        for col in cat_cols:
            if df[col].isnull().sum() > 0:
                df[col] = df[col].fillna('Missing')
        
        return df

2. Robust Model Training with Time-Aware Validation

Innovation: Custom time-series cross-validation strategy that respects temporal order and prevents data leakage, crucial for financial fraud detection.


import lightgbm as lgb
from sklearn.metrics import roc_auc_score, precision_recall_curve
import warnings
warnings.filterwarnings('ignore')

class FraudDetectionModel:
    def __init__(self):
        self.models = {}
        self.feature_importance = {}
    
    def time_series_cv(self, X, y, n_splits=5):
        """Time-aware cross-validation respecting transaction order"""
        tscv = TimeSeriesSplit(n_splits=n_splits)
        cv_scores = []
        feature_importance = []
        
        for fold, (train_idx, val_idx) in enumerate(tscv.split(X)):
            print(f"Training fold {fold + 1}/{n_splits}")
            
            X_train, X_val = X.iloc[train_idx], X.iloc[val_idx]
            y_train, y_val = y.iloc[train_idx], y.iloc[val_idx]
            
            # Train LightGBM with class weights for imbalance
            model = lgb.LGBMClassifier(
                n_estimators=1000,
                learning_rate=0.05,
                num_leaves=31,
                class_weight='balanced',
                random_state=42
            )
            
            model.fit(
                X_train, y_train,
                eval_set=[(X_val, y_val)],
                eval_metric='auc',
                early_stopping_rounds=100,
                verbose=False
            )
            
            # Validate and store results
            val_pred = model.predict_proba(X_val)[:, 1]
            auc_score = roc_auc_score(y_val, val_pred)
            cv_scores.append(auc_score)
            
            # Store feature importance
            fold_importance = pd.DataFrame({
                'feature': X.columns,
                'importance': model.feature_importances_,
                'fold': fold
            })
            feature_importance.append(fold_importance)
            
            self.models[f'fold_{fold}'] = model
        
        return cv_scores, pd.concat(feature_importance)
    
    def optimize_threshold(self, X_val, y_val):
        """Find optimal classification threshold for precision-recall tradeoff"""
        val_pred = self.predict_proba(X_val)
        
        precision, recall, thresholds = precision_recall_curve(y_val, val_pred)
        f1_scores = 2 * (precision * recall) / (precision + recall + 1e-8)
        
        # Find threshold that maximizes F1-score
        optimal_idx = np.argmax(f1_scores[:-1])  # exclude last element
        optimal_threshold = thresholds[optimal_idx]
        
        return optimal_threshold

3. Explainable AI with SHAP Integration

Innovation: Custom SHAP explanation engine that provides both global feature importance and local instance explanations, bridging the gap between model predictions and business understanding.


import shap
import matplotlib.pyplot as plt
import seaborn as sns

class SHAPExplainer:
    def __init__(self, model, feature_names):
        self.model = model
        self.feature_names = feature_names
        self.explainer = None
        self.shap_values = None
    
    def compute_global_explanations(self, X, sample_size=10000):
        """Compute global SHAP values for feature importance"""
        # Sample data for faster computation
        if len(X) > sample_size:
            X_sample = X.sample(sample_size, random_state=42)
        else:
            X_sample = X
        
        # Initialize TreeExplainer for LightGBM
        self.explainer = shap.TreeExplainer(self.model)
        self.shap_values = self.explainer.shap_values(X_sample)
        
        return self.shap_values
    
    def generate_summary_plot(self, X_sample, max_display=20):
        """Create SHAP summary plot for global feature importance"""
        plt.figure(figsize=(10, 8))
        
        # For binary classification, use SHAP values for class 1 (fraud)
        if isinstance(self.shap_values, list):
            shap_array = self.shap_values[1]  # Fraud class
        else:
            shap_array = self.shap_values
        
        shap.summary_plot(
            shap_array, 
            X_sample,
            feature_names=self.feature_names,
            max_display=max_display,
            show=False
        )
        
        plt.tight_layout()
        plt.savefig('shap_summary.png', dpi=300, bbox_inches='tight')
        plt.close()
    
    def explain_single_prediction(self, instance):
        """Generate local explanation for a single transaction"""
        # Get SHAP values for this instance
        instance_shap = self.explainer.shap_values(instance)
        
        if isinstance(instance_shap, list):
            instance_shap = instance_shap[1]  # Fraud class
        
        # Create force plot for individual prediction
        shap.force_plot(
            self.explainer.expected_value[1] if isinstance(self.explainer.expected_value, list) 
            else self.explainer.expected_value,
            instance_shap,
            instance,
            feature_names=self.feature_names,
            matplotlib=True,
            show=False
        )
        
        plt.tight_layout()
        plt.savefig('single_prediction_explanation.png', dpi=300, bbox_inches='tight')
        plt.close()
        
        return instance_shap
    
    def get_top_features(self, n_features=10):
        """Extract top features driving fraud predictions"""
        if isinstance(self.shap_values, list):
            shap_array = self.shap_values[1]
        else:
            shap_array = self.shap_values
        
        # Calculate mean absolute SHAP values
        feature_importance = pd.DataFrame({
            'feature': self.feature_names,
            'importance': np.abs(shap_array).mean(0)
        })
        
        return feature_importance.nlargest(n_features, 'importance')
